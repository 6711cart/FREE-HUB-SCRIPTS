local function createToggle(name, defaultState, yPos, callback)
    local btn = Instance.new("TextButton")
    btn.Name = name .. "Button"
    btn.Size = UDim2.new(0.9, 0, 0, 32)
    btn.Position = UDim2.new(0.05, 0, 0, yPos)
    btn.BackgroundColor3 = Color3.fromRGB(24, 34, 52)
    btn.Font = Enum.Font.GothamSemibold
    btn.TextSize = 12
    btn.ZIndex = 7
    btn.Parent = ScrollContainer

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = btn

    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromRGB(50, 90, 140)
    stroke.Transparency = 0.6
    stroke.Thickness = 1
    stroke.Parent = btn

    local state = defaultState
    local function updateText()
        if state then
            btn.Text = "  " .. name .. ": [ ON ]"
            btn.TextColor3 = Color3.fromRGB(120, 255, 160)
        else
            btn.Text = "  " .. name .. ": [ OFF ]"
            btn.TextColor3 = Color3.fromRGB(255, 120, 120)
        end
    end
    updateText()

    btn.MouseButton1Click:Connect(function()
        state = not state
        updateText()
        callback(state)
    end)
end

-- Chams (Outline + Inside Rainbow Fill) & 3D Billboard Name Tags Handler
local highlights = {}
local billBoards = {}

local function removeESP(player)
    if highlights[player] then
        highlights[player]:Destroy()
        highlights[player] = nil
    end
    if billBoards[player] then
        billBoards[player]:Destroy()
        billBoards[player] = nil
    end
end

Players.PlayerRemoving:Connect(function(player)
    removeESP(player)
end)

local function isVisible(targetPart)
    if not Config.WallCheck then return true end
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return true end
    
    local origin = Camera.CFrame.Position
    local direction = (targetPart.Position - origin)
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = {char, Camera}
    raycastParams.IgnoreWater = true
    
    local result = workspace:Raycast(origin, direction, raycastParams)
    if result then
        if result.Instance:IsDescendantOf(targetPart.Parent) then
            return true
        end
        return false
    end
    return true
end

local function getClosestPlayerInFOV()
    local closestPlayer = nil
    local shortestDistance = math.huge
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
            local targetPart = player.Character:FindFirstChild(Config.TargetPart)
            
            if humanoid and humanoid.Health > 0 and targetPart and isVisible(targetPart) then
                local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                if onScreen then
                    local magnitude = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                    if magnitude < shortestDistance then
                        shortestDistance = magnitude
                        closestPlayer = targetPart
                    end
                end
            end
        end
    end
    return closestPlayer
end

local function applyNoRecoil()
    if not Config.NoRecoil then return end
    pcall(function()
        local char = LocalPlayer.Character
        if char then
            for _, obj in ipairs(char:GetChildren()) do
                if obj:IsA("Tool") then
                    for _, v in ipairs(obj:GetDescendants()) do
                        if v:IsA("NumberValue") or v:IsA("IntValue") then
                            if v.Name:lower():find("recoil") or v.Name:lower():find("spread") or v.Name:lower():find("kick") then
                                v.Value = 0
                            end
                        end
                    end
                end
            end
        end
    end)
end

-- Main Render Loop
local rotationAngle = 0
RunService:BindToRenderStep("FrostbiteUrbanUpdate", Enum.RenderPriority.Camera.Value + 1, function(dt)
    pcall(function()
        local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        local tickTime = tick() * 2
        local rainbowColor = Color3.fromHSV((tickTime % 5) / 5, 1, 1)

        applyNoRecoil()

        -- Update Chams & Attached 3D Billboard Name Tags for Everyone (Including LocalPlayer)
        for _, player in ipairs(Players:GetPlayers()) do
            local char = player.Character
            local humanoid = char and char:FindFirstChildOfClass("Humanoid")
            local head = char and char:FindFirstChild("Head")

            if char and humanoid and humanoid.Health > 0 then
                -- Chams Outline & Inside Rainbow Fill (Visible on everyone including self)
                if Config.ChamsESP then
                    local hl = highlights[player]
                    if not hl then
                        hl = Instance.new("Highlight")
                        hl.Name = "FrostbiteChams"
                        hl.Adornee = char
                        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        hl.FillTransparency = 0.5 -- Inside fill transparency (rainbow color inside)
                        hl.OutlineTransparency = 0
                        hl.Parent = CoreGui
                        highlights[player] = hl
                    else
                        hl.Adornee = char
                    end
                    hl.OutlineColor = rainbowColor
                    hl.FillColor = rainbowColor
                else
                    if highlights[player] then
                        highlights[player]:Destroy()
                        highlights[player] = nil
                    end
                end

                -- Attached 3D Billboard Name Tags (Locked to Head, won't drift)
                if Config.NameTags and head then
                    local bb = billBoards[player]
                    local textLabel
                    if not bb then
                        bb = Instance.new("BillboardGui")
                        bb.Name = "FrostbiteNameTag"
                        bb.Size = UDim2.new(0, 100, 0, 40)
                        bb.StudsOffset = Vector3.new(0, 2.2, 0)
                        bb.AlwaysOnTop = true
                        bb.Adornee = head

                        textLabel = Instance.new("TextLabel")
                        textLabel.Name = "Label"
                        textLabel.Size = UDim2.new(1, 0, 1, 0)
                        textLabel.BackgroundTransparency = 1
                        textLabel.Font = Enum.Font.GothamBold
                        textLabel.TextSize = 14
                        textLabel.TextStrokeTransparency = 0
                        textLabel.Parent = bb

                        bb.Parent = CoreGui
                        billBoards[player] = bb
                    else
                        bb.Adornee = head
                        textLabel = bb:FindFirstChild("Label")
                    end

                    if textLabel then
                        textLabel.Text = player.Name
                        textLabel.TextColor3 = rainbowColor
                    end
                else
                    if billBoards[player] then
                        billBoards[player]:Destroy()
                        billBoards[player] = nil
                    end
                end
            else
                removeESP(player)
            end
        end

        -- Crosshair Rendering (Default: Swastika)
        if Config.CustomCrosshair then
            if Config.Spinning then
                rotationAngle = rotationAngle + (Config.SpinSpeed * dt)
            end

            for _, line in ipairs(crosshairLines) do
                line.Visible = false
            end

            if Config.CrosshairStyle == "Swastika" then
                local size = 16
                local radAngle = math.rad(rotationAngle)
                local cosA = math.cos(radAngle)
                local sinA = math.sin(radAngle)

                local function rotatePoint(pt)
                    return screenCenter + Vector2.new(pt.X * cosA - pt.Y * sinA, pt.X * sinA + pt.Y * cosA)
                end

                local segments = {
                    {Vector2.new(0, 0), Vector2.new(0, -size)},
                    {Vector2.new(0, -size), Vector2.new(size, -size)},
                    {Vector2.new(0, 0), Vector2.new(size, 0)},
                    {Vector2.new(size, 0), Vector2.new(size, size)},
                    {Vector2.new(0, 0), Vector2.new(0, size)},
                    {Vector2.new(0, size), Vector2.new(-size, size)},
                    {Vector2.new(0, 0), Vector2.new(-size, 0)},
                    {Vector2.new(-size, 0), Vector2.new(-size, -size)}
                }

                for i, lineData in ipairs(segments) do
                    local line = crosshairLines[i]
                    if line then
                        line.Color = rainbowColor
                        line.Visible = true
                        line.From = rotatePoint(lineData[1])
                        line.To = rotatePoint(lineData[2])
                    end
                end
            elseif Config.CrosshairStyle == "Star" then
                local outerRadius = 18
                local innerRadius = 7.5
                local points = {}
                
                for i = 0, 9 do
                    local radius = (i % 2 == 0) and outerRadius or innerRadius
                    local angle = math.rad(rotationAngle + (i * 36) - 90)
                    table.insert(points, screenCenter + Vector2.new(math.cos(angle), math.sin(angle)) * radius)
                end
                
                for i = 1, 10 do
                    local line = crosshairLines[i]
                    if line then
                        line.Color = rainbowColor
                        line.Visible = true
                        line.From = points[i]
                        line.To = points[(i % 10) + 1]
                    end
                end
            end
        else
            for _, line in ipairs(crosshairLines) do
                line.Visible = false
            end
        end

        -- Aimbot Logic (Works only when scoping/holding Right Mouse Button)
        if Config.Enabled then
            local isScoping = not Config.AutoScopeActivate or UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
            if isScoping then
                local target = getClosestPlayerInFOV()
                if target then
                    Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, target.Position)
                end
            end
        end
    end)
end)
